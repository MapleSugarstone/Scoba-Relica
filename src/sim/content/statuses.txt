# Every status a move or a passive can leave on a Scoba.
# The language this file is written in is described in docs/move-script.md.

status ez "EZ Mode"
  good
  stacks
  always as written
  while carried:
    all stats +3

status fire "Fire"
  bad
  icon firework
  lasts 3 turns
  stacks
  when a turn ends:
    damage holder 15% of source magic, as firework magic, fixed when applied

status fragile "Fragile"
  bad
  icon broken_bone
  lasts 2 turns
  lost on switching out
  while carried:
    defense x0.9
    resistance x0.9

status rage "Rage"
  good
  stacks up to 6
  lost on switching out
  while carried:
    strength x1.25

status guard "Guard"
  good
  lasts 3 turns
  lost on switching out
  while carried:
    defense x1.25

status moonward "Moonward"
  good
  icon moon
  lasts 2 turns
  while carried:
    immune to moon

status marked "Marked"
  bad
  lasts 3 turns
  while carried:
    takes x1.5 from cipher

status second-wind "Second Wind"
  good
  charges 1
  when below 50% hp:
    heal holder 10% of their max hp

status sticky "Sticky Treat"
  bad
  icon sticky
  sound stickysweet
  lasts 1 turn
  while carried:
    cannot switch out

status slowed "Slowed"
  bad
  icon boot
  stacks up to 6
  lost on switching out
  power 30% of source magic
  while carried:
    speed - power

status cold "Cold"
  bad
  icon cold
  lasts 3 turns

    inflict chill on holder

status chill "Chill"
  bad

    speed - power

status hyper "Hyper-Mode"

status deathlock "Deathlock"
  good
  icon timemachine
  lasts 3 turns
  when it faints:
    show deathclock as clock over holder, pointer deathclockhand
    flash #fff4dd for 0.5 seconds
    rewind 3 turns
    clear deathlock from holder
    inflict unwound on holder
    holder wears spent
    give holder time-shatter in slot 4

status undoing "Undoing"
  good
  text "At the end of the round, the damage and statuses taken while this held are put back."
  icon undoing
  lasts 1 turn

status unwound "Unwound"
  bad
  while carried:
    all stats x0.8
    cannot enter hyper-mode

# The sigils Multiply Enemies and Multiply Allies show on the Scobas they reach.
# Nothing carries these: they are shown for as long as the reach lasts, with a
# stack for each Scoba on the field reaching it.
status multiply-enemies-reach "Multiply Enemies"
  bad
  icon givingmultiply
  text "Each enemy on the field with Multiply Enemies makes this Scoba's negative statuses 15% more effective."

status multiply-allies-reach "Multiply Allies"
  good
  icon multiplygood
  text "Each ally on the field with Multiply Allies makes this Scoba's positive statuses 15% more effective."

# Plib and Poba. The guard it answers a blow with, the strength it works up,
# and the dent its attacks leave in whoever it is swinging at.

  icon selfdefense
  stacks up to 4
  lost on switching out
  spends a stack
  text "Answers physical damage with a free basic attack on the enemy that dealt it. Each answer spends a stack."

    basic attack other
